#include <Arduino.h>
#include <esp_system.h>

// ============================================================
// CONFIGURACIÓN
// ============================================================

const uint32_t BAUDRATE_SERIAL = 115200;
const uint32_t INTERVALO_ENVIO_MS = 500;

uint32_t tiempo_anterior_envio = 0;


// ============================================================
// INICIALIZAR SERIAL
// ============================================================

void inicializarSerial(uint32_t baudrate)
{
    Serial.begin(baudrate);

    while (!Serial)
    {
        delay(10);
    }

    randomSeed((uint32_t)esp_random());

    Serial.println("{\"estado\":\"serial_iniciado\"}");
}


// ============================================================
// GENERAR VALORES ALEATORIOS
// ============================================================

float generarSensor1()
{
    // Valor float entre 0.000 y 100.000
    return random(0, 100001) / 1000.0;
}


int generarSensor2()
{
    // Valor entero entre 0 y 500
    return random(0, 501);
}


int generarSensor3()
{
    // Valor tipo byte entre 0 y 255
    return random(0, 256);
}


// ============================================================
// ENVIAR DATO JSON
// ============================================================

void enviarDato()
{
    float sensor_1 = generarSensor1();
    int sensor_2 = generarSensor2();
    int sensor_3 = generarSensor3();

    Serial.print("{");
    Serial.print("\"sensor_1\":");
    Serial.print(sensor_1, 3);
    Serial.print(",");

    Serial.print("\"sensor_2\":");
    Serial.print(sensor_2);
    Serial.print(",");

    Serial.print("\"sensor_3\":");
    Serial.print(sensor_3);

    Serial.println("}");
}


// ============================================================
// ENVÍO PERIÓDICO
// ============================================================

void ejecutarEnvioPeriodico()
{
    uint32_t tiempo_actual = millis();

    if (tiempo_actual - tiempo_anterior_envio >= INTERVALO_ENVIO_MS)
    {
        tiempo_anterior_envio = tiempo_actual;
        enviarDato();
    }
}


// ============================================================
// SETUP
// ============================================================

void setup()
{
    inicializarSerial(BAUDRATE_SERIAL);
}


// ============================================================
// LOOP
// ============================================================

void loop()
{
    ejecutarEnvioPeriodico();

    // Aquí en el futuro podrás meter más lógica sin bloquear:
    // - leer sensores reales
    // - gestionar estados
    // - recibir comandos por serial
}
